import { Client, Account } from 'node-aacsearch';
import { H3Event, getCookie } from 'h3';

async function getLoggedInUser(event) {
  const session = getCookie(event, 'custom-session-cookie');
  if (!session) return;

  const client = new Client()
    .setEndpoint(process.env.PUBLIC_AACSEARCH_ENDPOINT)
    .setProject(process.env.PUBLIC_AACSEARCH_PROJECT_ID);

  client.setSession(session.value);
  const account = new Account(client);

  return account.get();
}
